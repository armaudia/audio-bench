cmake_minimum_required(VERSION 3.20)
project(audio_bench_native VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
if(NOT AUDIO_SDK_PREFIX)
  message(FATAL_ERROR "Set AUDIO_SDK_PREFIX to the verified installed SDK prefix")
endif()
execute_process(
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_sdk_snapshot.py" --prefix "${AUDIO_SDK_PREFIX}"
  RESULT_VARIABLE SDK_VERIFY_RESULT
  OUTPUT_VARIABLE SDK_VERIFY_OUTPUT
  ERROR_VARIABLE SDK_VERIFY_ERROR
)
if(NOT SDK_VERIFY_RESULT EQUAL 0)
  message(FATAL_ERROR "Pinned SDK verification failed: ${SDK_VERIFY_OUTPUT}${SDK_VERIFY_ERROR}")
endif()
message(STATUS "${SDK_VERIFY_OUTPUT}")
file(REAL_PATH "${AUDIO_SDK_PREFIX}" AUDIO_SDK_PREFIX_REAL)
set(AudioSDK_DIR "${AUDIO_SDK_PREFIX_REAL}/lib/cmake/AudioSDK")
find_package(AudioSDK 0.1 CONFIG REQUIRED
  PATHS "${AudioSDK_DIR}"
  NO_DEFAULT_PATH
  NO_CMAKE_PACKAGE_REGISTRY
  NO_CMAKE_SYSTEM_PACKAGE_REGISTRY)
if(NOT TARGET AudioSDK::audio_sdk)
  message(FATAL_ERROR "Installed package lacks exported AudioSDK::audio_sdk target")
endif()
if(NOT AudioSDK_CONFIG)
  message(FATAL_ERROR "CMake did not report the selected AudioSDKConfig.cmake")
endif()
file(REAL_PATH "${AudioSDK_CONFIG}" AUDIO_SDK_CONFIG_FILE_REAL)
file(REAL_PATH "${AUDIO_SDK_PREFIX_REAL}/lib/cmake/AudioSDK/AudioSDKConfig.cmake"
  EXPECTED_AUDIO_SDK_CONFIG_REAL)
if(NOT AUDIO_SDK_CONFIG_FILE_REAL STREQUAL EXPECTED_AUDIO_SDK_CONFIG_REAL)
  message(FATAL_ERROR "AudioSDKConfig.cmake resolved outside verified AUDIO_SDK_PREFIX")
endif()
get_target_property(AUDIO_SDK_INCLUDES AudioSDK::audio_sdk INTERFACE_INCLUDE_DIRECTORIES)
foreach(include_dir IN LISTS AUDIO_SDK_INCLUDES)
  file(REAL_PATH "${include_dir}" include_real)
  if(NOT include_real STREQUAL "${AUDIO_SDK_PREFIX_REAL}/include")
    message(FATAL_ERROR "AudioSDK imported headers resolve outside verified prefix: ${include_real}")
  endif()
endforeach()
get_target_property(AUDIO_SDK_CONFIGS AudioSDK::audio_sdk IMPORTED_CONFIGURATIONS)
foreach(config IN LISTS AUDIO_SDK_CONFIGS)
  get_target_property(imported_location AudioSDK::audio_sdk "IMPORTED_LOCATION_${config}")
  if(imported_location)
    file(REAL_PATH "${imported_location}" imported_real)
    cmake_path(IS_PREFIX AUDIO_SDK_PREFIX_REAL "${imported_real}" NORMALIZE imported_is_pinned)
    if(NOT imported_is_pinned)
      message(FATAL_ERROR "AudioSDK imported library resolves outside verified prefix: ${imported_real}")
    endif()
  endif()
endforeach()
file(READ "${CMAKE_CURRENT_LIST_DIR}/../deps/lock.json" BENCH_DEPENDENCY_LOCK)
string(JSON DEPENDENCY_COUNT LENGTH "${BENCH_DEPENDENCY_LOCK}" dependencies)
set(AUDIO_BENCH_SDK_PACKAGE_SHA256 "")
math(EXPR DEPENDENCY_LAST "${DEPENDENCY_COUNT} - 1")
foreach(index RANGE ${DEPENDENCY_LAST})
  string(JSON dependency_id GET "${BENCH_DEPENDENCY_LOCK}" dependencies ${index} id)
  if(dependency_id STREQUAL "audio-sdk-installed-package")
    string(JSON AUDIO_BENCH_SDK_PACKAGE_SHA256 GET "${BENCH_DEPENDENCY_LOCK}" dependencies ${index} sha256)
    string(REGEX REPLACE "^sha256:" "" AUDIO_BENCH_SDK_PACKAGE_SHA256 "${AUDIO_BENCH_SDK_PACKAGE_SHA256}")
  endif()
endforeach()
string(LENGTH "${AUDIO_BENCH_SDK_PACKAGE_SHA256}" AUDIO_BENCH_SDK_PACKAGE_SHA256_LENGTH)
if(NOT AUDIO_BENCH_SDK_PACKAGE_SHA256_LENGTH EQUAL 64 OR
   NOT AUDIO_BENCH_SDK_PACKAGE_SHA256 MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "Could not resolve exact SDK archive digest from deps/lock.json")
endif()
add_executable(audio_bench_driver src/driver.cpp src/wire.cpp)
target_include_directories(audio_bench_driver PRIVATE src)
target_link_libraries(audio_bench_driver PRIVATE AudioSDK::audio_sdk ${CMAKE_DL_LIBS})
target_compile_definitions(audio_bench_driver PRIVATE AUDIO_BENCH_SDK_PACKAGE_SHA256="${AUDIO_BENCH_SDK_PACKAGE_SHA256}")
add_executable(audio_bench_collect src/collect.cpp src/wire.cpp)
target_include_directories(audio_bench_collect PRIVATE src)
target_link_libraries(audio_bench_collect PRIVATE AudioSDK::audio_sdk)
target_compile_definitions(audio_bench_collect PRIVATE AUDIO_BENCH_SDK_PACKAGE_SHA256="${AUDIO_BENCH_SDK_PACKAGE_SHA256}")
add_executable(audio_bench_d256_regression tests/d256_regression.cpp tests/d256_provider.cpp)
target_include_directories(audio_bench_d256_regression PRIVATE tests)
target_link_libraries(audio_bench_d256_regression PRIVATE AudioSDK::audio_sdk)
add_library(audio_bench_test_provider MODULE tests/provider_plugin.cpp)
target_link_libraries(audio_bench_test_provider PRIVATE AudioSDK::audio_sdk)
set_target_properties(audio_bench_test_provider PROPERTIES PREFIX "" OUTPUT_NAME "audio_bench_test_provider")
add_library(audio_bench_asset_test_provider MODULE tests/provider_asset_plugin.cpp)
target_link_libraries(audio_bench_asset_test_provider PRIVATE AudioSDK::audio_sdk)
set_target_properties(audio_bench_asset_test_provider PROPERTIES PREFIX "" OUTPUT_NAME "audio_bench_asset_test_provider")
foreach(target audio_bench_driver audio_bench_collect audio_bench_d256_regression audio_bench_test_provider audio_bench_asset_test_provider)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_test(NAME sdk_snapshot_reverify COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../scripts/verify_sdk_snapshot.py" --prefix "${AUDIO_SDK_PREFIX}")
add_test(NAME sdk_d256_startup_validity_regression
  COMMAND audio_bench_d256_regression
    "${CMAKE_CURRENT_LIST_DIR}/../tests/fixtures/d256/descriptor.json"
    "${CMAKE_CURRENT_LIST_DIR}/../tests/fixtures/d256/config-schema.json")
